cmake_minimum_required(VERSION 3.16)
project(LibraryOOP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

add_library(core STATIC src/LibraryItem.cpp src/Member.cpp src/Library.cpp)
target_include_directories(core PUBLIC src)

add_executable(library_console src/main_console.cpp)
target_link_libraries(library_console core)

find_package(Qt6 COMPONENTS Widgets QUIET)
if(Qt6_FOUND)
  set(QT_LIBS Qt6::Widgets)
else()
  find_package(Qt5 COMPONENTS Widgets)
  set(QT_LIBS Qt5::Widgets)
endif()
if(Qt6_FOUND OR Qt5_FOUND)
  add_executable(library_gui src/main_gui.cpp src/MainWindow.cpp src/MainWindow.h src/Style.h)
  target_link_libraries(library_gui core ${QT_LIBS})
else()
  message(WARNING "Qt not found: only library_console is built")
endif()
